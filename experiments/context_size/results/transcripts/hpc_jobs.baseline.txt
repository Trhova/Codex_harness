# HPC Jobs Fixture

This fixture mimics a small repository of Slurm job scripts. It has shell utilities, job entry points, and operational notes for queue and checkpoint behavior.

--- docs/operations.md ---
# Operations Notes

Use the GPU partition only for training jobs. CPU preprocessing should run as a Slurm array so failed shards can be retried independently.

Checkpoints are written under `checkpoints/` and logs are written under `logs/`. When a queue is busy, submit preprocessing first and start training with an `afterok` dependency.

--- jobs/array_preprocess.sbatch ---
#!/usr/bin/env bash
#SBATCH --job-name=preprocess-array
#SBATCH --array=0-15
#SBATCH --partition=cpu
#SBATCH --mem=8G
#SBATCH --time=01:00:00

set -euo pipefail
source scripts/common.sh

load_modules python/3.11
prepare_workspace "${SLURM_ARRAY_TASK_ID}"
python preprocess.py --shard "${SLURM_ARRAY_TASK_ID}" --shards 16

--- scripts/common.sh ---
#!/usr/bin/env bash

load_modules() {
  for module_name in "$@"; do
    module load "${module_name}"
  done
}

prepare_workspace() {
  local run_id="$1"
  mkdir -p "runs/${run_id}" checkpoints logs
  export TMPDIR="runs/${run_id}/tmp"
  mkdir -p "${TMPDIR}"
}

submit_with_dependency() {
  local dependency="$1"
  local script="$2"
  sbatch --dependency="afterok:${dependency}" "${script}"
}

--- scripts/monitor.sh ---
#!/usr/bin/env bash
set -euo pipefail

queue="${1:-$USER}"
squeue --user "${queue}" --format="%.18i %.9P %.24j %.8T %.10M %.6D %R"

if compgen -G "logs/*.err" > /dev/null; then
  grep -n "OutOfMemory\\|CANCELLED\\|checkpoint" logs/*.err || true
fi
